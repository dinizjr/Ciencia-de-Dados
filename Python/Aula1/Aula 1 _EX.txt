    if a<b:
        if b<c:
            return c
        else:
            return b
    else:
        if a<c:
            return c
        else:
            return a

	def even_odd(numbers):
    numbers.sort()
    odd = []
    even = []

    for number in numbers:
        if(number % 2 == 0):
            even.append(number)
        else:
            odd.append(number)

    ret = (even, odd)
    return ret

def intervalSum(x,y):
    sum = 0
    if x > y:
        z = x
        x = y
        y = z

    for i in range(x, y + 1):
        sum = sum + i

    return sum

print(intervalSum(9, 9))

def fatorialWhile(n):
    i = 1
    n_fat = 1

    while i <= n:
        n_fat = n_fat * i
        i = i + 1

    return n_fat

def fatorialFor(n):
    n_fat = 1

    for i in range(2, n + 1):
        n_fat = n_fat * i

    return n_fat


print(fatorialWhile(0))
print(fatorialWhile(1))
print(fatorialWhile(5))
print(fatorialFor(0))
print(fatorialFor(1))
print(fatorialFor(5))
